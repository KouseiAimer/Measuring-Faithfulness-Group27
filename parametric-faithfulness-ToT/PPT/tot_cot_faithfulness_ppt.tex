\begin{frame}{Parametric Faithfulness: CoT vs. ToT}
\small
\begin{columns}[T,onlytextwidth]
  \begin{column}{0.53\textwidth}
    \textbf{Question}

    \vspace{0.35em}
    Does multi-path Tree-of-Thought reasoning expose a stronger dependence
    on the model's internal parameters than a single Chain-of-Thought path?

    \vspace{0.9em}
    \textbf{Core idea}
    \begin{itemize}
      \item Generate reasoning traces for multiple-choice science QA.
      \item Unlearn selected reasoning steps while preserving held-out behavior.
      \item Re-evaluate whether the original answer flips after the intervention.
    \end{itemize}
  \end{column}

  \begin{column}{0.43\textwidth}
    \textbf{Runs completed}

    \vspace{0.35em}
    \begin{tabularx}{\linewidth}{X r r}
      \toprule
      Dataset & Forget & Retain \\
      \midrule
      OpenBookQA & 50 & 20 \\
      ARC-Challenge & 100 & 20 \\
      \bottomrule
    \end{tabularx}

    \vspace{0.9em}
    \textbf{Model and update}
    \begin{itemize}
      \item Llama-3.2-3B-Instruct
      \item NPO-KL unlearning
      \item FF2-only updates, 5 epochs
      \item same hyperparameters for CoT and ToT
    \end{itemize}
  \end{column}
\end{columns}
\end{frame}

      \midrule
      OpenBookQA & sample-select & 80.0 & 5.00 \\
      ARC-Challenge & sample-select & 80.0 & 5.00 \\

    \end{tabularx}
  \end{column}

  \begin{column}{0.47\textwidth}
    \textbf{Metrics}

    \vspace{0.35em}
    \begin{itemize}
      \item \metric{Eff}: removal of the unlearned reasoning behavior.
      \item \metric{Spec}: preservation on retained examples.
      \item \metric{FF-HARD}: answer flips after unlearning among eligible questions.
      \item \metric{FF-SOFT}: probability shift away from the original answer.
      \item \metric{Post-CoT Agree}: post-reasoning answer agreement filter.
    \end{itemize}

    \vspace{0.6em}
    \begin{block}{Integrity check}
      All four main runs completed with full question coverage. No malformed
      JSON, duplicate steps, missing epochs, tracebacks, or CUDA OOMs were found.
    \end{block}
  \end{column}
\end{columns}
\end{frame}

\begin{frame}{Main Results}
\small
\begin{columns}[T,onlytextwidth]
  \begin{column}{0.61\textwidth}
    \centering
    \scriptsize
    \renewcommand{\arraystretch}{1.15}
    \begin{tabularx}{\linewidth}{X l r r r}
      \toprule
      Dataset & Arm & Eff & Spec & FF-HARD \\
      \midrule
      OpenBookQA & CoT & 81.2 & 95.4 & 72.5 \\
      OpenBookQA & ToT-selected & \gain{81.8} & 95.1 & 71.8 \\
      \midrule
      ARC-Challenge & CoT & 75.8 & \gain{95.6} & 68.8 \\
      ARC-Challenge & ToT-selected & \gain{79.6} & 95.4 & \gain{72.2} \\
      \bottomrule
    \end{tabularx}

    \vspace{0.7em}
    \scriptsize
    \begin{tabularx}{\linewidth}{X r r r}
      \toprule
      Dataset & $\Delta$Eff & $\Delta$FF-HARD & $\Delta$FF-SOFT \\
      \midrule
      OpenBookQA & \gain{+0.6} & \loss{-0.7} & \gain{+5.2} \\
      ARC-Challenge & \gain{+3.8} & \gain{+3.4} & \gain{+7.3} \\
      \bottomrule
    \end{tabularx}
  \end{column}

  \begin{column}{0.35\textwidth}
    \textbf{Takeaways}
    \begin{itemize}
      \item ARC-Challenge shows the clearest ToT advantage:
      higher Eff, FF-HARD, and FF-SOFT with almost unchanged Spec.
      \item OpenBookQA improves in soft probability shift, but not in
      strict paper-protocol FF-HARD.
      \item Paired eligible subset: ARC gains \(+3.2\) FF-HARD points;
      OpenBookQA is tied.
    \end{itemize}

    \vspace{0.4em}
    \scriptsize
    MMLU Gen was not run in this batch, so it is not imputed from other metrics.
  \end{column}
\end{columns}
\end{frame}
